{$CLEO .cs}
// SA · AW FPS  —  Loadout: three exo classes, key 8 / 9 / 0
// Compile: Sanny Builder, GTA SA → CLEO\AW_FPS_Loadout.cs
// F4 disables class swaps so you keep mission guns when the pack is off.
// Pressing the same class again is a no-op (does not strip ammo).

0000: NOP

int 0@
int 1@
int 2@
int 3@

0@ = 0
1@ = 0
2@ = 1
3@ = 0

while true
    wait 0

    if
        0AB0:   key_pressed 0x73
    then
        if
            3@ == 0
        then
            if
                2@ == 1
            then
                2@ = 0
            else
                2@ = 1
            end
        end
        3@ = 1
    else
        3@ = 0
    end

    if
        2@ == 0
    then
        jump @load_idle
    end

    if
        0256:   player $PLAYER_CHAR defined
    then
        if and
            044B:   actor $PLAYER_ACTOR on_foot
            03EE:   player $PLAYER_CHAR controllable
        then
            if
                0AB0:   key_pressed 0x38
            then
                if
                    1@ == 0
                then
                    if
                        8039:   not  0@ == 1
                    then
                        gosub @give_assault
                    end
                    1@ = 1
                end
            else
                if
                    0AB0:   key_pressed 0x39
                then
                    if
                        1@ == 0
                    then
                        if
                            8039:   not  0@ == 2
                        then
                            gosub @give_heavy
                        end
                        1@ = 1
                    end
                else
                    if
                        0AB0:   key_pressed 0x30
                    then
                        if
                            1@ == 0
                        then
                            if
                                8039:   not  0@ == 3
                            then
                                gosub @give_spec
                            end
                            1@ = 1
                        end
                    else
                        1@ = 0
                    end
                end
            end
        end
    end

:load_idle
end

:strip_class_guns
0555: remove_weapon_from_actor $PLAYER_ACTOR 16
0555: remove_weapon_from_actor $PLAYER_ACTOR 17
0555: remove_weapon_from_actor $PLAYER_ACTOR 24
0555: remove_weapon_from_actor $PLAYER_ACTOR 25
0555: remove_weapon_from_actor $PLAYER_ACTOR 29
0555: remove_weapon_from_actor $PLAYER_ACTOR 31
0555: remove_weapon_from_actor $PLAYER_ACTOR 34
0555: remove_weapon_from_actor $PLAYER_ACTOR 35
0555: remove_weapon_from_actor $PLAYER_ACTOR 38
return

:give_assault
0@ = 1
gosub @strip_class_guns
01B2: give_actor $PLAYER_ACTOR weapon 31 ammo 300
01B2: give_actor $PLAYER_ACTOR weapon 24 ammo 70
01B2: give_actor $PLAYER_ACTOR weapon 16 ammo 3
01B9: set_actor $PLAYER_ACTOR armed_weapon_to 31
0223: set_actor $PLAYER_ACTOR health_to 100
0ACA: show_text_box "CLASS  ASSAULT"
return

:give_heavy
0@ = 2
gosub @strip_class_guns
01B2: give_actor $PLAYER_ACTOR weapon 25 ammo 40
01B2: give_actor $PLAYER_ACTOR weapon 35 ammo 4
01B2: give_actor $PLAYER_ACTOR weapon 16 ammo 5
01B9: set_actor $PLAYER_ACTOR armed_weapon_to 25
0223: set_actor $PLAYER_ACTOR health_to 150
0ACA: show_text_box "CLASS  HEAVY"
return

:give_spec
0@ = 3
gosub @strip_class_guns
01B2: give_actor $PLAYER_ACTOR weapon 34 ammo 30
01B2: give_actor $PLAYER_ACTOR weapon 29 ammo 200
01B2: give_actor $PLAYER_ACTOR weapon 17 ammo 4
01B9: set_actor $PLAYER_ACTOR armed_weapon_to 34
0223: set_actor $PLAYER_ACTOR health_to 80
0ACA: show_text_box "CLASS  SPECIALIST"
return
