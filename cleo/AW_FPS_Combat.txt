{$CLEO .cs}
// SA · AW FPS  —  Combat: hitmarker, medals, streak rewards
// Compile: Sanny Builder, GTA SA → CLEO\AW_FPS_Combat.cs
// Stock CLEO 4.4: 0AE1 scan instead of CLEO+ 0AB3.

0000: NOP

int 0@
int 1@
int 2@
int 3@
int 4@
int 5@
int 6@
int 7@
int 8@
int 9@
int 13@
float 10@
float 11@
float 12@

0@ = 0          // unused score bucket (toasts still say +50)
1@ = 0          // streak count
2@ = 0          // streak timeout ms
3@ = 0          // hitmarker frames
4@ = 0          // tracked actor
5@ = 0          // tracked health
6@ = 0          // firing
9@ = 0          // last killed handle (debounce)

while true
    wait 0

    if
        0256:   player $PLAYER_CHAR defined
    then
        if and
            044B:   actor $PLAYER_ACTOR on_foot
            03EE:   player $PLAYER_CHAR controllable
        then
            6@ = 0
            if or
                00E1:   player 0 pressed_key 17
                0AB0:   key_pressed 0x01
            then
                6@ = 1
            end

            if
                3@ > 0
            then
                3@ -= 1
            end
            if
                2@ > 0
            then
                2@ -= 16
            else
                1@ = 0
            end

            04C4: store_coords_from_actor $PLAYER_ACTOR into 10@ 11@ 12@

            8@ = 0
            :scan_actors
            0AE1: 7@ = random_actor_near_point 10@ 11@ 12@ in_radius 16.0 find_next 8@ pass_deads 1
            if
                7@ == -1
            then
                jump @scan_done
            end
            if
                803B:   not  7@ == $PLAYER_ACTOR
            then
                if
                    8118:   not actor 7@ dead
                then
                    0226: 13@ = actor 7@ health
                    if
                        4@ == 7@
                    then
                        if
                            13@ < 5@
                        then
                            if
                                6@ == 1
                            then
                                3@ = 8
                                if
                                    13@ <= 1
                                then
                                    if
                                        803B:   not  7@ == 9@
                                    then
                                        9@ = 7@
                                        gosub @register_kill
                                    end
                                end
                            end
                        end
                    end
                    4@ = 7@
                    5@ = 13@
                else
                    if
                        6@ == 1
                    then
                        if
                            803B:   not  7@ == 9@
                        then
                            9@ = 7@
                            gosub @register_kill
                        end
                    end
                    if
                        4@ == 7@
                    then
                        4@ = 0
                    end
                end
            end
            8@ = 1
            jump @scan_actors
            :scan_done

            if
                3@ > 0
            then
                03F0: enable_text_draw 1
                038E: draw_box_position 318.0 224.0 size 6.0 2.0 RGBA 255 255 255 220
                038E: draw_box_position 322.0 224.0 size 6.0 2.0 RGBA 255 255 255 220
                038E: draw_box_position 320.0 222.0 size 2.0 6.0 RGBA 255 255 255 220
                038E: draw_box_position 320.0 226.0 size 2.0 6.0 RGBA 255 255 255 220
            end
        end
    end
end

:register_kill
1@ += 1
2@ = 8000
3@ = 10
0ACA: show_text_box "ELIMINATED  +50"

if
    1@ == 2
then
    0ACA: show_text_box "DOUBLE KILL"
end
if
    1@ == 3
then
    0ACA: show_text_box "TRIPLE KILL"
end
if
    1@ == 5
then
    0ACA: show_text_box "RAMPAGE"
end
if
    1@ == 20
then
    01B2: give_actor $PLAYER_ACTOR weapon 35 ammo 4
    0ACA: show_text_box "STREAK 20  —  RPG"
end
if
    1@ == 30
then
    01B2: give_actor $PLAYER_ACTOR weapon 38 ammo 200
    0ACA: show_text_box "STREAK 30  —  DEATH MACHINE"
end
return
